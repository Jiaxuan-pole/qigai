import { wishPressure, mindStep, diseaseStep, makeDowned, resolveDeadline, foodReserve, validateDialogue, validateEventProposal } from './public/game/rules.js';
const { rules } = await loadData();
let actor = { mind: 28, health: 82, zeroTurns: 0, crisis: false };

while (actor.health > 0 && turn < 30) { actor = mindStep(actor, wishPressure([100, 100, 100], rules), 0, rules); turn++; }

const deadline = makeDowned('xuan', turn, false, rules).deadline;

let disease = { severity: 70, health: 100 };
for (let i = 0; i < 10; i++) disease = diseaseStep(disease.severity, disease.health, { matchingCare: true, rest: true }, rules);
assert.equal(disease.severity, 0, '真实导入运行十次护理必须清零病情');
const tail = makeDowned('ma', rules.durationDays * rules.slotsPerDay, false, rules);

assert.equal(validateDialogue(dialogue, context).ok, true, '正常解析的对话必须通过');
dialogue.choiceLabels[0].choiceId = ['rest'];
assert.equal(validateDialogue(dialogue, context).ok, false, '解析后的数组选项标识必须拒绝');
